epoch_core_migrate_planetary_features = {
	every_owned_planet = {
		if = {
			limit = { OR = { has_modifier = epoch_core_contamination_active_organic has_modifier = epoch_core_contamination_active_hive has_modifier = epoch_core_contamination_active_machine has_modifier = epoch_core_contamination_worsened_organic has_modifier = epoch_core_contamination_worsened_hive has_modifier = epoch_core_contamination_worsened_machine has_modifier = epoch_core_contamination_studied_organic has_modifier = epoch_core_contamination_studied_hive has_modifier = epoch_core_contamination_studied_machine has_modifier = epoch_core_contamination_active has_planet_flag = epoch_core_contamination_spread } }
			if = {
			limit = { OR = { has_modifier = epoch_core_contamination_worsened_organic has_modifier = epoch_core_contamination_worsened_hive has_modifier = epoch_core_contamination_worsened_machine has_planet_flag = epoch_core_contamination_spread root = { OR = { has_country_flag = epoch_core_contamination_worsened has_country_flag = epoch_core_contamination_neglected has_country_flag = epoch_core_contamination_study_failed has_country_flag = epoch_core_contamination_spread_recorded } } } }
				epoch_core_apply_contamination_worsened = yes
			}
			else_if = {
				limit = { OR = { has_modifier = epoch_core_contamination_studied_organic has_modifier = epoch_core_contamination_studied_hive has_modifier = epoch_core_contamination_studied_machine root = { has_country_flag = epoch_core_contamination_study_complete } } }
				epoch_core_apply_contamination_studied = yes
			}
			else_if = {
				limit = { root = { has_country_flag = epoch_core_contamination_contained } }
	epoch_core_remove_contamination_effects = yes
				set_planet_flag = epoch_core_contamination_contained_state
				add_deposit = d_epoch_resonance_contamination_contained
			}
			else = { epoch_core_apply_contamination_active = yes }
		}
		if = {
			limit = { OR = { has_modifier = epoch_core_permanent_scar_organic has_modifier = epoch_core_permanent_scar_hive has_modifier = epoch_core_permanent_scar_machine } }
			epoch_core_set_resonance_scar = yes
		}
		if = {
			limit = { OR = { has_modifier = epoch_core_silent_structure has_modifier = epoch_core_structure_secured has_modifier = epoch_core_structure_investigated has_modifier = epoch_core_structure_ruins } NOR = { has_deposit = d_epoch_silent_structure_unsecured has_deposit = d_epoch_silent_structure_secured has_deposit = d_epoch_silent_structure_investigated has_deposit = d_epoch_silent_structure_destroyed } }
			if = { limit = { root = { has_country_flag = epoch_core_structure_destroyed } } epoch_core_set_structure_destroyed = yes }
			else_if = { limit = { root = { has_country_flag = epoch_core_structure_secured } } epoch_core_set_structure_secured = yes }
			else_if = { limit = { root = { has_country_flag = epoch_core_structure_investigated } } epoch_core_set_structure_investigated = yes }
			else = { epoch_core_set_structure_unsecured = yes }
		}
		if = {
			limit = { OR = { has_modifier = epoch_ii_resonance_latent has_modifier = epoch_ii_resonance_active has_modifier = epoch_ii_resonance_unstable has_modifier = epoch_ii_resonance_controlled has_planet_flag = epoch_ii_latent_resonance has_planet_flag = epoch_ii_active_resonance has_planet_flag = epoch_ii_unstable_resonance has_planet_flag = epoch_ii_controlled_resonance } }
			remove_modifier = epoch_ii_resonance_latent
			remove_modifier = epoch_ii_resonance_active
			remove_modifier = epoch_ii_resonance_unstable
			remove_modifier = epoch_ii_resonance_controlled
			remove_deposit = d_epoch_resonance_contamination_active
			remove_deposit = d_epoch_resonance_contamination_worsened
			remove_deposit = d_epoch_resonance_contamination_studied
			remove_deposit = d_epoch_resonance_contamination_contained
			remove_deposit = d_epoch_resonance_contamination_latent
			remove_deposit = d_epoch_resonance_contamination_resonant
			remove_deposit = d_epoch_resonance_contamination_unstable
			remove_deposit = d_epoch_resonance_contamination_controlled
			if = { limit = { has_planet_flag = epoch_ii_latent_resonance } add_deposit = d_epoch_resonance_contamination_latent }
			else_if = { limit = { has_planet_flag = epoch_ii_active_resonance } add_deposit = d_epoch_resonance_contamination_resonant }
			else_if = { limit = { has_planet_flag = epoch_ii_unstable_resonance } add_deposit = d_epoch_resonance_contamination_unstable }
			else = { add_deposit = d_epoch_resonance_contamination_controlled }
		}
	}
}

# Keep chapter targets inside the campaign owner's current territory. A lost
# Phase-I world falls back to the capital or another owned planet.
epoch_ii_refresh_campaign_target = {
	if = {
		limit = { OR = { NOT = { exists = event_target:epoch_ii_target_planet@root } event_target:epoch_ii_target_planet@root = { NOT = { is_owned_by = root } } } }
		if = {
			limit = { exists = capital_scope }
			capital_scope = { save_global_event_target_as = epoch_ii_target_planet@root solar_system = { save_global_event_target_as = epoch_ii_target_system@root } }
			remove_country_flag = epoch_ii_no_owned_campaign_planet
		}
		else_if = {
			limit = { any_owned_planet = { always = yes } }
			random_owned_planet = { save_global_event_target_as = epoch_ii_target_planet@root solar_system = { save_global_event_target_as = epoch_ii_target_system@root } }
			remove_country_flag = epoch_ii_no_owned_campaign_planet
		}
		else = { set_country_flag = epoch_ii_no_owned_campaign_planet }
	}
}

# A resolved withdrawal or successful signal lure de-escalates only the
# campaign owner's relation and orders the living fleet toward empty space.
# The fleet remains in the galaxy and can still be fought by the player.
epoch_ii_disengage_nullwarden_fleet = {
	if = {
		limit = { has_country_flag = epoch_ii_nullwarden_disengage_complete }
		set_country_flag = epoch_ii_nullwarden_disengage_complete
	}
	else_if = {
		limit = { exists = event_target:epoch_ii_nullwarden_fleet@root }
		event_target:epoch_ii_nullwarden_fleet@root = { owner = { save_global_event_target_as = epoch_ii_nullwarden_country@root } }
		if = {
			limit = { NOT = { has_country_flag = epoch_ii_nullwarden_disengage_issued } }
			event_target:epoch_ii_nullwarden_fleet@root = {
				owner = { set_faction_hostility = { target = root set_hostile = no set_neutral = yes } }
				clear_orders = yes
				set_fleet_stance = evasive
			}
			set_country_flag = epoch_ii_nullwarden_disengage_issued
			set_country_flag = epoch_ii_nullwarden_retreat_pending
		}
		if = {
			limit = {
				has_country_flag = epoch_ii_nullwarden_route_queued
				exists = event_target:epoch_ii_retreat_system@root
				event_target:epoch_ii_retreat_system@root = { has_access_fleet = event_target:epoch_ii_nullwarden_country@root }
			}
			event_target:epoch_ii_nullwarden_fleet@root = {
				if = {
					limit = { solar_system = { is_same_value = event_target:epoch_ii_retreat_system@root } }
					set_fleet_stance = passive
					root = { remove_country_flag = epoch_ii_nullwarden_retreat_pending remove_country_flag = epoch_ii_nullwarden_route_queued remove_country_flag = epoch_ii_retreat_destination_found set_country_flag = epoch_ii_nullwarden_disengage_complete }
				}
			}
		}
		else = {
			remove_country_flag = epoch_ii_nullwarden_route_queued
			remove_country_flag = epoch_ii_retreat_destination_found
			event_target:epoch_ii_nullwarden_fleet@root = {
				set_fleet_stance = evasive
				random_system = {
					limit = {
						has_owner = no
						has_access_fleet = event_target:epoch_ii_nullwarden_country@root
						distance = { source = prev.solar_system use_bypasses = no min_jumps = 2 max_jumps = 4 }
						NOT = { is_same_value = event_target:epoch_ii_target_system@root }
					}
					save_global_event_target_as = epoch_ii_retreat_system@root
					root = { set_country_flag = epoch_ii_retreat_destination_found }
				}
				if = {
					limit = { root = { NOT = { has_country_flag = epoch_ii_retreat_destination_found } } }
					closest_system = {
						limit = {
							has_owner = no
							has_access_fleet = event_target:epoch_ii_nullwarden_country@root
							NOT = { is_same_value = prev.solar_system }
							NOT = { is_same_value = event_target:epoch_ii_target_system@root }
						}
						save_global_event_target_as = epoch_ii_retreat_system@root
						root = { set_country_flag = epoch_ii_retreat_destination_found }
					}
				}
				if = {
					limit = { root = { NOT = { has_country_flag = epoch_ii_retreat_destination_found } } }
					closest_system = {
						limit = {
							has_access_fleet = event_target:epoch_ii_nullwarden_country@root
							NOT = { is_same_value = prev.solar_system }
							NOT = { is_same_value = event_target:epoch_ii_target_system@root }
						}
						save_global_event_target_as = epoch_ii_retreat_system@root
						root = { set_country_flag = epoch_ii_retreat_destination_found }
					}
				}
				if = {
					limit = { root = { has_country_flag = epoch_ii_retreat_destination_found } }
					queue_actions = { move_to = event_target:epoch_ii_retreat_system@root }
					root = { set_country_flag = epoch_ii_nullwarden_route_queued set_country_flag = epoch_ii_nullwarden_retreat_pending }
				}
				else = { set_fleet_stance = passive root = { set_country_flag = epoch_ii_nullwarden_retreat_pending } }
			}
		}
	}
	else = {
		remove_country_flag = epoch_ii_nullwarden_retreat_pending
		remove_country_flag = epoch_ii_nullwarden_route_queued
		remove_country_flag = epoch_ii_retreat_destination_found
		set_country_flag = epoch_ii_nullwarden_disengage_complete
	}
}

# The Chapter 6 fleet has its own saved target and idempotency flags, but uses
# the same neutralize-then-retreat policy as the Block A force.
epoch_ii_disengage_second_nullwarden_fleet = {
	if = {
		limit = { exists = event_target:epoch_ii_second_fleet@root }
		event_target:epoch_ii_second_fleet@root = { owner = { save_global_event_target_as = epoch_ii_second_nullwarden_country@root } }
		if = {
			limit = { NOT = { has_country_flag = epoch_ii_second_force_disengage_issued } }
			event_target:epoch_ii_second_fleet@root = {
				owner = { set_faction_hostility = { target = root set_hostile = no set_neutral = yes } }
				clear_orders = yes
				set_fleet_stance = evasive
			}
			set_country_flag = epoch_ii_second_force_disengage_issued
			set_country_flag = epoch_ii_second_force_retreat_pending
		}
		if = {
			limit = { has_country_flag = epoch_ii_second_force_route_queued exists = event_target:epoch_ii_second_retreat_system@root event_target:epoch_ii_second_retreat_system@root = { has_access_fleet = event_target:epoch_ii_second_nullwarden_country@root } }
			event_target:epoch_ii_second_fleet@root = {
				if = {
					limit = { solar_system = { is_same_value = event_target:epoch_ii_second_retreat_system@root } }
					set_fleet_stance = passive
					root = { remove_country_flag = epoch_ii_second_force_retreat_pending remove_country_flag = epoch_ii_second_force_route_queued remove_country_flag = epoch_ii_second_force_no_safe_route set_country_flag = epoch_ii_second_force_disengage_complete }
				}
			}
		}
		else = {
			remove_country_flag = epoch_ii_second_force_route_queued
			remove_country_flag = epoch_ii_second_retreat_destination_found
			event_target:epoch_ii_second_fleet@root = {
				set_fleet_stance = evasive
				random_system = {
					limit = { has_owner = no has_access_fleet = event_target:epoch_ii_second_nullwarden_country@root distance = { source = prev.solar_system use_bypasses = no min_jumps = 2 max_jumps = 4 } NOT = { is_same_value = event_target:epoch_ii_target_system@root } }
					save_global_event_target_as = epoch_ii_second_retreat_system@root
					root = { set_country_flag = epoch_ii_second_retreat_destination_found }
				}
				if = {
					limit = { root = { NOT = { has_country_flag = epoch_ii_second_retreat_destination_found } } }
					closest_system = {
						limit = { has_access_fleet = event_target:epoch_ii_second_nullwarden_country@root NOT = { is_same_value = prev.solar_system } NOT = { is_same_value = event_target:epoch_ii_target_system@root } }
						save_global_event_target_as = epoch_ii_second_retreat_system@root
						root = { set_country_flag = epoch_ii_second_retreat_destination_found }
					}
				}
				if = {
					limit = { root = { has_country_flag = epoch_ii_second_retreat_destination_found } }
					queue_actions = { move_to = event_target:epoch_ii_second_retreat_system@root }
					root = { remove_country_flag = epoch_ii_second_force_no_safe_route set_country_flag = epoch_ii_second_force_route_queued set_country_flag = epoch_ii_second_force_retreat_pending }
				}
				else = { set_fleet_stance = evasive root = { set_country_flag = epoch_ii_second_force_no_safe_route set_country_flag = epoch_ii_second_force_retreat_pending } }
			}
		}
	}
	else = {
		remove_country_flag = epoch_ii_second_force_retreat_pending
		remove_country_flag = epoch_ii_second_force_route_queued
		remove_country_flag = epoch_ii_second_force_no_safe_route
		set_country_flag = epoch_ii_second_force_disengage_complete
	}
}
