# Global coordinator scope. Each call covers one recorded Stage III front.
epoch_frontiers_reconcile_ruin = {
 if = {
  limit = {
   has_global_flag = epoch_iii_front_$FRONT$_destroyed
   exists = event_target:epoch_iii_front_$FRONT$
  }
  event_target:epoch_iii_front_$FRONT$ = {
   if = {
    limit = { exists = star NOT = { has_star_flag = epoch_frontiers_ruin } }
    star = {
     create_ambient_object = { type = derelict_ship_object location = this entity_offset = { min = 18 max = 18 } }
    }
    set_star_flag = epoch_frontiers_ruin
   }
   if = {
    limit = {
     has_star_flag = epoch_frontiers_ruin
     NOT = { has_star_flag = epoch_frontiers_surveyed }
     exists = starbase
     starbase = {
      epoch_frontiers_station_operational = yes
      owner = { NOT = { has_special_project = $PROJECT$ } }
     }
    }
    starbase = { owner = { save_event_target_as = epoch_frontiers_survey_owner } }
    star = {
     enable_special_project = {
      name = $PROJECT$
      location = this
      owner = event_target:epoch_frontiers_survey_owner
     }
    }
   }
  }
 }
}
