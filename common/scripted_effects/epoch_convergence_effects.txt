# Only the canonical hostile country writes this saved chapter variable.
# Global flags below are compatibility facts, never per-country progression.
epoch_iii_update_chapter = {
 if = { limit = { is_country_type = epoch_convergence has_country_flag = epoch_iii_identity_v1 has_global_flag = epoch_iii_active }
  if = { limit = { NOT = { has_global_flag = epoch_iii_chapters_v2 } }
   set_global_flag = epoch_iii_chapters_v2
   set_variable = { which = epoch_iii_chapter value = 1 }
   if = { limit = { has_global_flag = epoch_iii_chapter_5 } set_variable = { which = epoch_iii_chapter value = 5 } }
   else_if = { limit = { has_global_flag = epoch_iii_command_created } set_variable = { which = epoch_iii_chapter value = 4 } }
   else_if = { limit = { has_global_flag = epoch_iii_chapter_3 } set_variable = { which = epoch_iii_chapter value = 3 } }
   else_if = { limit = { has_global_flag = epoch_iii_chapter_2 } set_variable = { which = epoch_iii_chapter value = 2 } }
  }
  # Reverse order plus dwell: at most one transition, even with banked kills.
  if = { limit = { NOT = { has_country_flag = epoch_iii_chapter_dwell } }
   if = { limit = { check_variable = { which = epoch_iii_chapter value = 4 } epoch_iii_all_nodes_destroyed = yes }
    set_variable = { which = epoch_iii_chapter value = 5 }
    set_global_flag = epoch_iii_chapter_5
    set_timed_country_flag = { flag = epoch_iii_chapter_dwell days = 90 }
    every_country = { limit = { has_country_flag = epoch_iii_enrolled } country_event = { id = epoch_iii.52 } }
   }
   else_if = { limit = { check_variable = { which = epoch_iii_chapter value = 3 } has_global_flag = epoch_iii_anchor_breached }
    set_variable = { which = epoch_iii_chapter value = 4 }
    set_global_flag = epoch_iii_chapter_4
    set_timed_country_flag = { flag = epoch_iii_chapter_dwell days = 90 }
    if = { limit = { NOT = { has_global_flag = epoch_iii_command_created } } epoch_iii_spawn_command = yes }
   }
   else_if = { limit = { check_variable = { which = epoch_iii_chapter value = 2 } has_global_flag = epoch_iii_battle_proven }
    set_variable = { which = epoch_iii_chapter value = 3 }
    set_global_flag = epoch_iii_chapter_3
    set_timed_country_flag = { flag = epoch_iii_chapter_dwell days = 90 }
    every_country = { limit = { has_country_flag = epoch_iii_enrolled } country_event = { id = epoch_iii.50 } }
   }
   else_if = { limit = { check_variable = { which = epoch_iii_chapter value = 1 } OR = { has_global_flag = epoch_iii_border_crossed has_global_flag = epoch_iii_battle_proven } }
    set_variable = { which = epoch_iii_chapter value = 2 }
    set_global_flag = epoch_iii_chapter_2
    set_timed_country_flag = { flag = epoch_iii_chapter_dwell days = 90 }
   }
  }
  every_country = { limit = { has_country_flag = epoch_iii_enrolled epoch_iii_public_empire = yes } epoch_iii_sync_display = yes }
  if = { limit = { has_global_flag = epoch_iii_chapter_5 }
   every_owned_fleet = { limit = { OR = { has_fleet_flag = epoch_iii_raid has_fleet_flag = epoch_iii_guard } NOT = { is_in_combat = yes } NOT = { is_disabled = yes } }
    epoch_iii_rally_fleet = yes
   }
  }
 }
}
