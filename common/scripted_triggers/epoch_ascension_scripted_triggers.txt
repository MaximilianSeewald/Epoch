  epoch_iv_eligible = yes

  any_country = {

    has_country_flag = epoch_core_campaign_active
    has_country_flag = epoch_res_active

   OR = {
    AND = {
     exists = event_target:epoch_core_world@this
     event_target:epoch_core_world@this = {

       is_same_value = prevprev
