# System scope. Never seize an asset or ambush a capital. A direct hyperlane
# into an ordinary non-capital system provides an accessible first front.
epoch_iii_safe_front = {
 NOT = { exists = owner }
 NOT = { exists = starbase }
 NOT = { any_fleet_in_system = { always = yes } }
 NOT = { any_system_planet = { is_colony = yes } }
 any_neighbor_system = {
  exists = owner
  owner = { epoch_iii_public_empire = yes }
  NOT = { is_capital_system = yes }
 }
}

# Saturated-map fallback: an uninhabited ordinary frontier outpost, never a
# fortified base or capital. Existing assets survive creation unchanged and
# can only be lost in the resulting native battle. No ownership transfer.
epoch_iii_frontier_outpost = {
 exists = owner
 owner = { epoch_iii_public_empire = yes }
 NOT = { is_capital_system = yes }
 NOT = { any_system_planet = { is_colony = yes } }
 exists = starbase
 starbase = { has_starbase_size = starbase_outpost }
 NOT = { any_fleet_in_system = { fleet_power > 500 } }
 any_neighbor_system = {
  exists = owner
  owner = { epoch_iii_public_empire = yes }
  NOT = { is_capital_system = yes }
 }
 any_neighbor_system = {
  OR = { NOT = { exists = owner } owner = { NOT = { is_same_value = prevprev.owner } } }
 }
}

# Spread fronts across one navigable hyperlane region; retain safe-asset rules.
epoch_iii_spaced_front = {
 NOT = { has_star_flag = epoch_iii_front_site }
 OR = { epoch_iii_safe_front = yes epoch_iii_frontier_outpost = yes }
 distance = { source = event_target:epoch_iii_front_1 type = hyperlane min_jumps = 4 max_jumps = 80 }

}
