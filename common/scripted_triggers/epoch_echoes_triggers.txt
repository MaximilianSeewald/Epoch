# Optional discoveries remain personal and only appear during the active early
# EPOCH inquiry. They never alter the Stage-III global crisis entry condition.
epoch_echoes_early_campaign = {
	epoch_core_eligible_empire = yes
	NOT = { has_country_flag = epoch_core_campaign_opted_out }
	OR = {
		has_country_flag = epoch_core_campaign_active
		has_country_flag = epoch_res_active
	}
	OR = {
		NOT = { has_country_flag = epoch_res_active }
		check_variable = { which = epoch_res_chapter value <= 6 }
	}
	OR = {
		NOT = { has_country_flag = epoch_core_campaign_active }
		check_variable = { which = epoch_core_current_phase value <= 2 }
	}
}

# Cheap guard: once all draws are negative or placed, opportunity handlers stop.
# Pending selection is independent for each story; no started-story exclusion.
epoch_echoes_pending_discovery = {
 OR = {
  NOT = { has_country_flag = epoch_echoes_ghost_rolled }
  NOT = { has_country_flag = epoch_echoes_unquiet_rolled }
  NOT = { has_country_flag = epoch_echoes_relay_rolled }
  AND = { has_country_flag = epoch_echoes_ghost_selected NOT = { has_country_flag = epoch_echoes_ghost_started } }
  AND = { has_country_flag = epoch_echoes_unquiet_selected NOT = { has_country_flag = epoch_echoes_unquiet_started } }
  AND = { has_country_flag = epoch_echoes_relay_selected NOT = { has_country_flag = epoch_echoes_relay_started } }
 }
}

# Planet scope; never reserve inhabited or protected campaign worlds.
epoch_echoes_space_site = {
 colonizable_planet = no
 is_star = no
 has_anomaly = no
 has_any_megastructure = no
 NOT = { has_planet_flag = has_megastructure }
 NOT = { has_planet_flag = epoch_echoes_reserved }
}

# Colony scope. The identity flag prevents conquest from transferring a story.
epoch_echoes_colony_owned = {
 exists = owner
 has_planet_flag = epoch_echoes_unquiet_for@owner
 owner = { NOT = { has_country_flag = epoch_core_campaign_opted_out } }
}

# Only main-campaign decisions tied to this colony take priority. Their
# visibility conditions still apply when an individual option is unaffordable.
epoch_echoes_colony_campaign_priority = {
 OR = {
  AND = {
   has_planet_flag = epoch_res_infected
   has_planet_flag = epoch_res_infected_for@owner
   owner = {
    has_country_flag = epoch_res_active
    NOT = { has_country_flag = epoch_core_campaign_opted_out }
    check_variable = { which = epoch_res_chapter value = 4 }
   }
  }
  AND = {
   epoch_iii_affected_world = yes
   OR = {
    NOT = { has_planet_flag = epoch_iii_buffered }
    NOT = { has_planet_flag = epoch_iii_treated }
   }
  }
 }
}

epoch_echoes_colony_strategy_available = {
 epoch_echoes_colony_owned = yes
 epoch_echoes_colony_campaign_priority = no
 has_planet_flag = epoch_echoes_unquiet_active
 NOT = { has_planet_flag = epoch_echoes_unquiet_closed }
 epoch_echoes_disturbance_present = yes
 NOT = { has_planet_flag = epoch_echoes_unquiet_treating }
}

epoch_echoes_disturbance_present = {
 OR = {
  has_modifier = epoch_echoes_unquiet_organic
  has_modifier = epoch_echoes_unquiet_hive
  has_modifier = epoch_echoes_unquiet_machine
 }
}


epoch_echoes_ghost_open = {
 has_country_flag = epoch_echoes_ghost_started
 NOT = { has_country_flag = epoch_echoes_ghost_closed }
 NOT = { has_country_flag = epoch_core_campaign_opted_out }
 exists = event_target:epoch_echoes_ghost_site@this
}
epoch_echoes_relay_open = {
 has_country_flag = epoch_echoes_relay_started
 NOT = { has_country_flag = epoch_echoes_relay_resolved }
 NOT = { has_country_flag = epoch_core_campaign_opted_out }
 exists = event_target:epoch_echoes_relay_object@this
 exists = event_target:epoch_echoes_relay_site@this
 event_target:epoch_echoes_relay_object@this = {
  is_megastructure_type = epoch_echoes_relay_dormant
  is_owned_by = prev
  has_megastructure_flag = epoch_echoes_relay_for@prev
 }
 event_target:epoch_echoes_relay_system@this = { is_inside_border = prev }
}
