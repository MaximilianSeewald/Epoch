# Optional discoveries remain personal and only appear during the active early
# EPOCH inquiry. They never alter the Stage-III global crisis entry condition.
epoch_echoes_early_campaign = {
	epoch_core_eligible_empire = yes
	NOT = { has_country_flag = epoch_core_campaign_opted_out }
	OR = {
		has_country_flag = epoch_core_campaign_active
		has_country_flag = epoch_res_active
	}
	OR = {
		NOT = { has_country_flag = epoch_res_active }
		check_variable = { which = epoch_res_chapter value <= 6 }
	}
	OR = {
		NOT = { has_country_flag = epoch_core_campaign_active }
		check_variable = { which = epoch_core_current_phase value <= 2 }
	}
}
