# Optional early-campaign side stories. Every project uses a staffed native
# Science Ship and persists at a surveyed, reachable map location.
special_project = {
	key = EPOCH_ECHOES_GHOST_TRACE
	days_to_research = 90
	tech_department = society_technology
	picture = GFX_evt_unknown_ships
	event_scope = ship_event
	location = yes
	requirements = { shipclass_science_ship = 1 leader = scientist }
	AI_wait_days = { weight = 5 }
	on_start = { if = { limit = { exists = owner exists = event_target:epoch_echoes_story_site@owner } event_target:epoch_echoes_story_site@owner = { save_global_event_target_as = epoch_echoes_ghost_trace_target@prev } } }
on_success = { if = { limit = { exists = owner exists = event_target:epoch_echoes_ghost_trace_target@this } ship_event = { id = epoch_echoes.40 scopes = { from = from fromfrom = event_target:epoch_echoes_ghost_trace_target@this } } } else = { country_event = { id = epoch_echoes.21 } } }
	on_fail = { country_event = { id = epoch_echoes.21 } }
	on_cancel = { country_event = { id = epoch_echoes.21 } }
}
