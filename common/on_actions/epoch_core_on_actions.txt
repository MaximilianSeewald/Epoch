# Country-local entry points keep each empire's bootstrap in its own scope.
on_game_start_country = {
	events = {
		epoch_core.1
	}
}

# Also catch empires created after the initial country-start dispatch.
on_country_created = {
	events = {
		epoch_core.1
	}
}

# Re-check an empire when it becomes a normal empire country type.
on_country_type_changed = {
	events = {
		epoch_core.1
	}
}

# Owner-local, guarded recovery/migration for continuing Phase-I saves.
on_yearly_pulse_country = {
	events = { epoch_core.51 epoch_ii.0 epoch_res.81 epoch_echoes.20 epoch_echoes.22 epoch_echoes.23 }
}

# The signal is attached to the system a Science Ship actually surveys.
on_survey_planet = {
	events = {
		epoch_core.15
		epoch_echoes.1
	}
}

# Native destruction callback: This=victim country, From=killer country,
# FromFrom=destroyed fleet. Participation hooks are fleet-scoped.
on_fleet_destroyed_victim = { events = { epoch_res.201 epoch_res.202 epoch_echoes.900 } }
on_fleet_combat_joined_attacker = { events = { epoch_res.200 } }
on_fleet_combat_joined_defender = { events = { epoch_res.200 } }
# This = each fleet leaving battle, whether it won, fled, or was destroyed.
on_fleet_exit_battle = { events = { epoch_res.203 } }

# Global load/monthly hooks have no country root. Scan only saved owner-local sites.
on_single_player_save_game_load = { events = { epoch_core.52 } }
on_monthly_pulse = { events = { epoch_core.52 } }
