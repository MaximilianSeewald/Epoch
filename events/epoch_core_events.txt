# Special-project callbacks run on the assigned Science Ship. The owner country
# advances its own Situation, so simultaneous empires never share progress.
ship_event = {
	id = epoch_core.15
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		owner = {
			epoch_core_eligible_empire = yes
			NOT = { has_country_flag = epoch_core_campaign_opted_out }
			NOT = { has_country_flag = epoch_core_first_echo_discovered }
			NOT = { has_country_flag = epoch_core_discovery_scheduled }
		}
	}
	immediate = {
		from = {
			solar_system = {
				save_global_event_target_as = epoch_core_expedition_origin_system@root.owner
				random_neighbor_system = {
					limit = { NOT = { exists = owner } }
					save_global_event_target_as = epoch_core_signal_system@root.owner
					random_system_planet = {
						save_global_event_target_as = epoch_core_expedition_destination@root.owner
					}
				}
			}
		}
		if = {
			limit = { NOT = { exists = event_target:epoch_core_signal_system@root.owner } }
			from = { solar_system = { save_global_event_target_as = epoch_core_signal_system@root.owner } }
		}
		if = {
			limit = { NOT = { exists = event_target:epoch_core_expedition_destination@root.owner } }
			from = { save_global_event_target_as = epoch_core_expedition_destination@root.owner }
		}
		owner = {
			set_country_flag = epoch_core_discovery_scheduled
			country_event = { id = epoch_core.2 days = 1 }
		}
	}
}

# The Situation's own monthly on_action dispatches this Situation-scoped event.
# It updates only feedback and one-shot approach risks; monthly_progress itself
# is the sole source of progress and no project is completed here.
situation_event = {
	id = epoch_core.47
	hide_window = yes
	is_triggered_only = yes
	situation = this
	trigger = { owner = { epoch_core_active_campaign_empire = yes } }
	immediate = {
		owner = {
			set_country_flag = epoch_core_monthly_dispatcher_seen
			country_event = { id = epoch_core.44 }
			country_event = { id = epoch_core.50 }
			remove_country_flag = epoch_core_situation_waiting_objective
			remove_country_flag = epoch_core_situation_waiting_progress
			remove_country_flag = epoch_core_situation_active_progress
		}
		if = {
			limit = { owner = { has_country_flag = epoch_core_situation_diagnostics } }
			log = "EPOCH_DIAG dispatcher=1 stage=[This.GetCurrentStage] progress=[This.GetProgress]"
			if = { limit = { owner = { has_country_flag = epoch_core_echo_first_project_pending } } log = "EPOCH_DIAG first_echo_project=unlocked" }
			else = { log = "EPOCH_DIAG first_echo_project=not_unlocked" }
			if = { limit = { owner = { OR = { has_country_flag = epoch_core_first_echo_complete has_country_flag = epoch_core_echo_second_abandoned } } } log = "EPOCH_DIAG first_echo_objective=complete" }
			else = { log = "EPOCH_DIAG first_echo_objective=open" }
		}
		if = {
			limit = {
				OR = {
				AND = { current_stage = stage_echo_resolution situation_progress >= 15 owner = { NOT = { OR = { has_country_flag = epoch_core_first_echo_complete has_country_flag = epoch_core_echo_second_abandoned } } } }
				AND = { current_stage = stage_expedition_resolution situation_progress >= 25 owner = { NOT = { has_country_flag = epoch_core_expedition_phase_resolved } } }
				AND = { current_stage = stage_contamination_resolution situation_progress >= 35 owner = { NOT = { has_country_flag = epoch_core_contamination_phase_resolved } } }
				AND = { current_stage = stage_structure_resolution situation_progress >= 45 owner = { NOT = { has_country_flag = epoch_core_structure_phase_resolved } } }
					AND = { current_stage = stage_conclusion_resolution situation_progress >= 55 owner = { NOT = { has_country_flag = epoch_core_conclusion_decision_resolved } } }
				}
			}
			owner = { set_country_flag = epoch_core_situation_waiting_objective }
		}
		else_if = {
			limit = {
				OR = {
					AND = { current_stage = stage_echo_resolution situation_progress < 20 owner = { OR = { has_country_flag = epoch_core_first_echo_complete has_country_flag = epoch_core_echo_second_abandoned } } }
					AND = { current_stage = stage_signal_analysis situation_progress < 15 }
					AND = { current_stage = stage_expedition_resolution situation_progress < 30 owner = { has_country_flag = epoch_core_expedition_phase_resolved } }
					AND = { current_stage = stage_contamination_resolution situation_progress < 40 owner = { has_country_flag = epoch_core_contamination_phase_resolved } }
					AND = { current_stage = stage_structure_resolution situation_progress < 50 owner = { has_country_flag = epoch_core_structure_phase_resolved } }
					AND = { current_stage = stage_conclusion_resolution situation_progress < 60 owner = { has_country_flag = epoch_core_conclusion_decision_resolved } }
				}
			}
			owner = { set_country_flag = epoch_core_situation_waiting_progress }
		}
		else = { owner = { set_country_flag = epoch_core_situation_active_progress } }
		if = {
			limit = {
				current_situation_approach = epoch_echo_containment_approach
				OR = { current_stage = stage_contamination current_stage = stage_contamination_resolution }
				owner = { NOT = { has_country_flag = epoch_core_contamination_phase_resolved } }
			}
			owner = {
				if = {
					limit = { exists = event_target:epoch_core_world@root.owner event_target:epoch_core_world@root.owner = { is_owned_by = root.owner } }

			}
		}
		else = { owner = { if = { limit = { exists = event_target:epoch_core_world@root.owner event_target:epoch_core_world@root.owner = { is_owned_by = root.owner } } event_target:epoch_core_world@root.owner = { remove_modifier = epoch_core_containment_buffered } } } }
		if = {
			limit = { current_stage = stage_signal_analysis current_situation_approach = epoch_echo_aggressive_approach }
			owner = { if = { limit = { NOT = { has_country_flag = epoch_core_intensive_risk_echo_checked } } set_country_flag = epoch_core_intensive_risk_echo_checked random_list = { 15 = { set_country_flag = epoch_core_signal_trace_degraded } 85 = { } } } }
		}
		if = {
			limit = { current_stage = stage_source_analysis current_situation_approach = epoch_echo_aggressive_approach }
			owner = { if = { limit = { NOT = { has_country_flag = epoch_core_intensive_risk_expedition_checked } } set_country_flag = epoch_core_intensive_risk_expedition_checked random_list = { 15 = { set_country_flag = epoch_core_signal_trace_degraded } 85 = { } } } }
		}
		if = {
			limit = { current_stage = stage_contamination_resolution current_situation_approach = epoch_echo_aggressive_approach owner = { NOT = { has_country_flag = epoch_core_contamination_phase_resolved } } }
			owner = {
				if = {
					limit = { NOT = { has_country_flag = epoch_core_intensive_risk_contamination_checked } exists = event_target:epoch_core_world@root.owner event_target:epoch_core_world@root.owner = { is_owned_by = root.owner } }
					set_country_flag = epoch_core_intensive_risk_contamination_checked
					random_list = {
						15 = {
							set_country_flag = epoch_core_contamination_worsened
						if = { limit = { exists = event_target:epoch_core_world@root.owner event_target:epoch_core_world@root.owner = { is_owned_by = root.owner } } event_target:epoch_core_world@root.owner = { epoch_core_remove_contamination_effects = yes epoch_core_apply_contamination_worsened = yes } }
						}
						85 = { }
					}
				}
			}
		}
		if = {
			limit = { current_stage = stage_excavation current_situation_approach = epoch_echo_aggressive_approach }
			owner = { if = { limit = { NOT = { has_country_flag = epoch_core_intensive_risk_structure_checked } } set_country_flag = epoch_core_intensive_risk_structure_checked random_list = { 20 = { set_country_flag = epoch_core_structure_risk_increased } 80 = { } } } }
		}
		if = {
			limit = { current_stage = stage_conclusion_resolution situation_progress >= 55 owner = { has_country_flag = epoch_core_conclusion_decision_resolved } }
			owner = {
				if = {
					limit = { NOT = { has_country_flag = epoch_core_phase_i_situation_complete } }
