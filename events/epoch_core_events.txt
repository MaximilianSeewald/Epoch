# Special-project callbacks run on the assigned Science Ship. The owner country
# advances its own Situation, so simultaneous empires never share progress.
ship_event = {
	id = epoch_core.15
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		owner = {
			epoch_core_eligible_empire = yes
			NOT = { has_country_flag = epoch_core_campaign_opted_out }
			NOT = { has_country_flag = epoch_core_first_echo_discovered }
			NOT = { has_country_flag = epoch_core_discovery_scheduled }
		}
	}
	immediate = {
		from = {
			solar_system = {
				random_neighbor_system = {
					limit = { NOT = { exists = owner } }
					save_global_event_target_as = epoch_core_signal_system@root.owner
				}
			}
		}
		if = {
			limit = { NOT = { exists = event_target:epoch_core_signal_system@root.owner } }
			from = { solar_system = { save_global_event_target_as = epoch_core_signal_system@root.owner } }
		}
		owner = {
			set_country_flag = epoch_core_discovery_scheduled
			country_event = { id = epoch_core.2 days = 1 }
		}
	}
}
