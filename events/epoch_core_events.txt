# Shared, idempotent entry point for starting empires and newly created ones.
# Campaign state: 0 = awaiting the opening choice, 1 = active, 2 = permanently
# opted out. campaign_exit_phase is 0 before a phase exit, or the phase number
# after which the empire left. A future phase-end exit must remove the active
# flag, set the permanent opt-out flag, set status 2, and record the current
# phase; old choice variables and flags remain intact. The First Echo is an
# introduction within Phase I, so this slice does not create a phase-end
# continue/exit prompt or placeholder content for Phase II.
country_event = {
	id = epoch_core.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		epoch_core_eligible_empire = yes
		NOT = { has_country_flag = epoch_core_campaign_opted_out }
		NOT = { has_country_flag = epoch_core_initialized }
	}

	immediate = {
		set_country_flag = epoch_core_initialized
		set_country_flag = epoch_core_discovery_scheduled
		set_variable = { which = epoch_core_schema_version value = 4 }
		set_variable = { which = epoch_core_discovery_status value = 0 }
		set_variable = { which = epoch_core_investigation_status value = 0 }
		set_variable = { which = epoch_core_completion_status value = 0 }
		set_variable = { which = epoch_core_campaign_status value = 0 }
		set_variable = { which = epoch_core_current_phase value = 0 }
		set_variable = { which = epoch_core_campaign_exit_phase value = 0 }
		set_variable = { which = epoch_core_phase_1_entry_decision value = 0 }
		set_variable = { which = epoch_core_expedition_status value = 0 }
		set_variable = { which = epoch_core_expedition_result value = 0 }
		country_event = { id = epoch_core.2 days = 30 }
	}
}
