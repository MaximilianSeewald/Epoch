country_event = {
 id = epoch_frontiers.2
 hide_window = yes
 is_triggered_only = yes
 trigger = {
  exists = from
  from = { has_star_flag = epoch_frontiers_surveyed }
 }
 immediate = {
  from = {
   remove_star_flag = epoch_frontiers_recovery_ready
   remove_star_flag = epoch_frontiers_wreck_one_ready
   remove_star_flag = epoch_frontiers_wreck_two_ready
   remove_star_flag = epoch_frontiers_recovered_for@root
   remove_star_flag = epoch_frontiers_recovery_notice
   remove_star_flag = epoch_frontiers_repair_notice
   if = {
    limit = { exists = starbase starbase = { epoch_frontiers_capacity_available = no } }
    set_star_flag = epoch_frontiers_damaged
   }
  }
 }
}

# The departing owner cannot finish a survey for the incoming owner.
ship_event = {
 id = epoch_frontiers.11
 hide_window = yes
 is_triggered_only = yes
 trigger = {
  has_global_flag = epoch_iii_first_repulsed
  exists = owner
  exists = leader
  exists = from
  from = { is_scope_type = country is_same_value = root.owner }
  exists = fromfrom
  solar_system = {
   is_same_value = fromfrom
   has_star_flag = epoch_frontiers_ruin
   NOT = { has_star_flag = epoch_frontiers_surveyed }
   exists = starbase
   starbase = {
    epoch_frontiers_station_operational = yes
    owner = { is_same_value = root.owner }
   }
  }
 }
 immediate = {
  solar_system = { set_star_flag = epoch_frontiers_surveyed }
  owner = { country_event = { id = epoch_frontiers.10 } }
 }
}

ship_event = {
 id = epoch_frontiers.20
 hide_window = yes
 is_triggered_only = yes
 trigger = {
  exists = owner
  exists = leader
  exists = from
  from = { is_scope_type = country is_same_value = root.owner }
  exists = fromfrom
  solar_system = {
   is_same_value = fromfrom
   epoch_frontiers_salvage_site = yes
   NOT = { has_star_flag = epoch_frontiers_recovery_ready }
   starbase = { owner = { is_same_value = root.owner } }
  }
 }
 immediate = {
  solar_system = {
   if = {
    limit = { has_star_flag = epoch_frontiers_wreck_one_spent }
    set_star_flag = epoch_frontiers_wreck_two_ready
   }
   else = { set_star_flag = epoch_frontiers_wreck_one_ready }
   set_star_flag = epoch_frontiers_recovery_ready
   set_star_flag = epoch_frontiers_recovered_for@root.owner
   epoch_frontiers_offer_recovery = yes
  }
 }
}
