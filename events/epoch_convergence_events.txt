# No-scope simulation coordinator. Durable reservation is independent of every
# empire's personal storyline, capital, survival or local UI choice.
event = {
 id = epoch_iii.1
 hide_window = yes
 is_triggered_only = yes
 immediate = {
  if = { limit = { NOT = { has_global_flag = epoch_iii_initialized } }
   set_global_flag = epoch_iii_initialized
   if = { limit = { mid_game_years_passed >= 0 }
    set_global_flag = epoch_iii_unsupported_late_install
    log = "EPOCH III: first initialization at/after midgame; automatic outbreak disabled (unsupported late installation)."
   }
   else = { set_global_flag = epoch_iii_schema_v1 }
  }
  if = { limit = { has_global_flag = epoch_iii_schema_v1 NOT = { has_global_flag = epoch_iii_unsupported_late_install } NOT = { has_global_flag = epoch_iii_first_repulsed } }
   if = { limit = { mid_game_years_passed >= -20 mid_game_years_passed < 0 }
    every_country = { limit = { epoch_iii_preparation_eligible = yes NOT = { has_country_flag = epoch_iii_warning_received } }
     set_country_flag = epoch_iii_warning_received
     country_event = { id = epoch_iii.5 }
    }
   }
   if = { limit = { mid_game_years_passed >= 0 }
    set_global_flag = epoch_iii_reserved
    if = { limit = { NOT = { has_global_flag = epoch_iii_country_created } }
     # One galaxy-wide selection per month only while placement is pending.
     random_system = { limit = { epoch_iii_safe_front = yes }
      save_global_event_target_as = epoch_iii_front
     }
     if = { limit = { NOT = { exists = event_target:epoch_iii_front } }
      random_system = { limit = { epoch_iii_frontier_outpost = yes }
       save_global_event_target_as = epoch_iii_front
      }
     }
     if = { limit = { exists = event_target:epoch_iii_front }
      set_global_flag = epoch_iii_country_created
      create_species = { name = NAME_EPOCH_III_Warden plural = NAME_EPOCH_III_Wardens class = ROBOT portrait = default_robot
       traits = { trait = trait_machine_unit } effect = { save_global_event_target_as = epoch_iii_species }
      }
      create_country = { species = event_target:epoch_iii_species name = NAME_EPOCH_III_Convergence type = epoch_convergence name_list = MACHINE1
       effect = {
        set_country_flag = epoch_iii_identity_v1
        set_variable = { which = epoch_iii_spent value = 0 }
        set_variable = { which = epoch_iii_raids_destroyed value = 0 }
        set_graphical_culture = fallen_machine_empire_01
        save_global_event_target_as = epoch_iii_country
       }
      }
     }
     else_if = { limit = { NOT = { has_global_flag = epoch_iii_placement_pending } }
      set_global_flag = epoch_iii_placement_pending
      log = "EPOCH III: no safe frontier; reserved outbreak retries monthly without spawning."
     }
    }
    # Identity recovery never creates another enemy or fabricates a death.
    if = { limit = { has_global_flag = epoch_iii_country_created NOT = { exists = event_target:epoch_iii_country } }
     every_country = { limit = { is_country_type = epoch_convergence has_country_flag = epoch_iii_identity_v1 }
      save_global_event_target_as = epoch_iii_country
     }
    }
    if = { limit = { exists = event_target:epoch_iii_country }
     event_target:epoch_iii_country = { country_event = { id = epoch_iii.2 } }
    }
    else_if = { limit = { has_global_flag = epoch_iii_country_created NOT = { has_global_flag = epoch_iii_reference_fault } }
     set_global_flag = epoch_iii_reference_fault
     log = "EPOCH III: canonical country absent; dispatch suspended, no victory or replacement granted."
    }
   }
  }
 }
}

# Fleet-root strategy. Local hyperlane search, one expensive path check per
# selected destination; 180-day replan covers stalled/lost/blocked orders.
fleet_event = {
 id = epoch_iii.30
 hide_window = yes
 is_triggered_only = yes
 trigger = { has_fleet_flag = epoch_iii_raid exists = owner owner = { is_country_type = epoch_convergence } NOT = { is_in_combat = yes } NOT = { is_disabled = yes } }
 immediate = {
  if = { limit = { has_global_flag = epoch_iii_chapter_5 }
   epoch_iii_rally_fleet = yes
  }
  else_if = { limit = { has_fleet_flag = epoch_iii_command } epoch_iii_command_move = yes }
  else_if = { limit = { OR = {
   NOT = { has_fleet_flag = epoch_iii_order_cooldown }
   NOT = { exists = event_target:epoch_iii_goal@this }
   AND = { exists = event_target:epoch_iii_goal@this event_target:epoch_iii_goal@this = { NOT = { exists = owner } } }
   AND = { exists = event_target:epoch_iii_goal@this solar_system = { is_same_value = event_target:epoch_iii_goal@root } }
  } }
   if = { limit = { exists = event_target:epoch_iii_goal@this } clear_global_event_target = epoch_iii_goal@this }
   solar_system = {
    if = { limit = { has_global_flag = epoch_iii_command_created }
     random_neighbor_system = { limit = { exists = owner owner = { epoch_iii_public_empire = yes } any_system_planet = { is_colony = yes } }
      save_global_event_target_as = epoch_iii_goal@root
     }
    }
    if = { limit = { NOT = { exists = event_target:epoch_iii_goal@root } }
    random_neighbor_system = {
     limit = { exists = owner owner = { epoch_iii_public_empire = yes } NOT = { is_capital_system = yes } }
     save_global_event_target_as = epoch_iii_goal@root
    }
    }
    if = { limit = { NOT = { exists = event_target:epoch_iii_goal@root } }
     random_neighbor_system = { limit = { any_neighbor_system = { exists = owner owner = { epoch_iii_public_empire = yes } } }
      random_neighbor_system = { limit = { exists = owner owner = { epoch_iii_public_empire = yes } }
       save_global_event_target_as = epoch_iii_goal@root
      }
     }
    }
    if = { limit = { NOT = { exists = event_target:epoch_iii_goal@root } }
     random_neighbor_system = { limit = { any_neighbor_system = { any_neighbor_system = { exists = owner owner = { epoch_iii_public_empire = yes } } } }
      random_neighbor_system = { limit = { any_neighbor_system = { exists = owner owner = { epoch_iii_public_empire = yes } } }
       random_neighbor_system = { limit = { exists = owner owner = { epoch_iii_public_empire = yes } }
        save_global_event_target_as = epoch_iii_goal@root
       }
      }
     }
    }
    if = { limit = { NOT = { exists = event_target:epoch_iii_goal@root } }
     random_neighbor_system = { limit = { NOT = { exists = owner } }
      save_global_event_target_as = epoch_iii_goal@root
     }
    }
   }
   set_timed_fleet_flag = { flag = epoch_iii_order_cooldown days = 180 }
   if = { limit = { exists = event_target:epoch_iii_goal@this can_access_system = event_target:epoch_iii_goal@this }
    clear_fleet_actions = this
    queue_actions = { move_to = event_target:epoch_iii_goal@this }
   }
  }
 }
}
