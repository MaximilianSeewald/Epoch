# Yearly owner-local recovery: flags guard invitation dispatch; deposit migration
# is idempotent and retries missing/foreign references. No global scan is used.
country_event = {
	id = epoch_ii.0
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_country_type = default
		has_country_flag = epoch_core_phase_i_concluded
		has_country_flag = epoch_core_phase_ii_ready
		NOT = { has_country_flag = epoch_core_campaign_opted_out }
	}
	immediate = {
		if = {
			limit = { NOT = { has_country_flag = epoch_ii_planet_migration_v2 } }
			# 0a226e1 treated missing and foreign references as final. Re-open
			# those records so this version can retry after ownership/reference changes.
			remove_country_flag = epoch_ii_contamination_migrated
			remove_country_flag = epoch_ii_capital_migrated
			remove_country_flag = epoch_ii_legacy_scar_features_migrated
			remove_country_flag = epoch_ii_contamination_feature_not_applicable
			remove_country_flag = epoch_ii_capital_feature_not_applicable
			remove_country_flag = epoch_ii_legacy_scar_feature_not_applicable
			remove_country_flag = epoch_ii_legacy_scar_feature_found
			remove_country_flag = epoch_ii_capital_legacy_feature_found
			set_country_flag = epoch_ii_planet_migration_v2
		}
		if = {
			limit = { NOT = { has_country_flag = epoch_ii_contamination_migrated } }
			if = {
				limit = { exists = event_target:epoch_core_world@root }
				if = {
					limit = { event_target:epoch_core_world@root = { is_owned_by = root } }
					event_target:epoch_core_world@root = {
						if = {
							limit = { root = { OR = { has_country_flag = epoch_core_contamination_worsened has_country_flag = epoch_core_contamination_neglected has_country_flag = epoch_core_contamination_study_failed has_country_flag = epoch_core_contamination_spread_recorded } } }
							if = { limit = { NOT = { has_deposit = d_epoch_contamination_worsened } } add_deposit = d_epoch_contamination_worsened }
						}
						else_if = { limit = { root = { has_country_flag = epoch_core_contamination_contained } } if = { limit = { NOT = { has_deposit = d_epoch_contamination_contained } } add_deposit = d_epoch_contamination_contained } }
						else_if = { limit = { root = { has_country_flag = epoch_core_contamination_study_complete } } if = { limit = { NOT = { has_deposit = d_epoch_contamination_studied } } add_deposit = d_epoch_contamination_studied } }
						else = { if = { limit = { NOT = { has_deposit = d_epoch_contamination_active } } add_deposit = d_epoch_contamination_active } }
						set_planet_flag = epoch_ii_contamination_feature_added
					}
					set_country_flag = epoch_ii_contamination_site_recorded
					set_country_flag = epoch_ii_contamination_migrated
					remove_country_flag = epoch_ii_contamination_feature_not_applicable
				}
				else = { set_country_flag = epoch_ii_contamination_feature_not_applicable }
			}
			else = { set_country_flag = epoch_ii_contamination_feature_not_applicable }
		}
		if = {
			limit = { NOT = { has_country_flag = epoch_ii_capital_migrated } }
			remove_country_flag = epoch_ii_capital_legacy_feature_found
			if = {
				limit = { exists = event_target:epoch_ii_conclusion_capital@root }
				if = {
					limit = { event_target:epoch_ii_conclusion_capital@root = { is_owned_by = root } }
					event_target:epoch_ii_conclusion_capital@root = {
						if = {
							limit = { has_modifier = epoch_core_permanent_scar_machine }
							if = { limit = { NOT = { has_deposit = d_epoch_conclusion_scar_machine } } add_deposit = d_epoch_conclusion_scar_machine }
							set_planet_flag = epoch_ii_conclusion_feature_added
							root = { set_country_flag = epoch_ii_capital_legacy_feature_found }
						}
						else_if = {
							limit = { has_modifier = epoch_core_permanent_scar_hive }
							if = { limit = { NOT = { has_deposit = d_epoch_conclusion_scar_hive } } add_deposit = d_epoch_conclusion_scar_hive }
							set_planet_flag = epoch_ii_conclusion_feature_added
							root = { set_country_flag = epoch_ii_capital_legacy_feature_found }
						}
						else_if = {
							limit = { has_modifier = epoch_core_permanent_scar_organic }
							if = { limit = { NOT = { has_deposit = d_epoch_conclusion_scar_organic } } add_deposit = d_epoch_conclusion_scar_organic }
							set_planet_flag = epoch_ii_conclusion_feature_added
							root = { set_country_flag = epoch_ii_capital_legacy_feature_found }
						}
					}
					if = { limit = { has_country_flag = epoch_ii_capital_legacy_feature_found } set_country_flag = epoch_ii_capital_migrated set_country_flag = epoch_ii_capital_scar_recorded remove_country_flag = epoch_ii_capital_feature_not_applicable }
					else = { set_country_flag = epoch_ii_capital_migrated set_country_flag = epoch_ii_capital_feature_not_applicable }
				}
				else = { set_country_flag = epoch_ii_capital_feature_not_applicable }
			}
			else = {
				# Legacy Phase-I saves lack the saved capital target. Search only
				# this empire's owned planets for the permanent scar marker.
				every_owned_planet = {
					if = { limit = { has_modifier = epoch_core_permanent_scar_machine } if = { limit = { NOT = { has_deposit = d_epoch_conclusion_scar_machine } } add_deposit = d_epoch_conclusion_scar_machine } set_planet_flag = epoch_ii_conclusion_feature_added root = { set_country_flag = epoch_ii_capital_legacy_feature_found } }
					else_if = { limit = { has_modifier = epoch_core_permanent_scar_hive } if = { limit = { NOT = { has_deposit = d_epoch_conclusion_scar_hive } } add_deposit = d_epoch_conclusion_scar_hive } set_planet_flag = epoch_ii_conclusion_feature_added root = { set_country_flag = epoch_ii_capital_legacy_feature_found } }
					else_if = { limit = { has_modifier = epoch_core_permanent_scar_organic } if = { limit = { NOT = { has_deposit = d_epoch_conclusion_scar_organic } } add_deposit = d_epoch_conclusion_scar_organic } set_planet_flag = epoch_ii_conclusion_feature_added root = { set_country_flag = epoch_ii_capital_legacy_feature_found } }
				}
				if = { limit = { has_country_flag = epoch_ii_capital_legacy_feature_found } set_country_flag = epoch_ii_capital_migrated set_country_flag = epoch_ii_capital_scar_recorded remove_country_flag = epoch_ii_capital_feature_not_applicable }
				else = { set_country_flag = epoch_ii_capital_feature_not_applicable }
			}
		}
		# Scan only this empire's current planets, every year. Deposits are
		# idempotent, and a previously foreign scar is picked up if reacquired.
		remove_country_flag = epoch_ii_legacy_scar_feature_found
		every_owned_planet = {
			if = { limit = { has_modifier = epoch_core_permanent_scar_organic } if = { limit = { NOT = { has_deposit = d_epoch_conclusion_scar_organic } } add_deposit = d_epoch_conclusion_scar_organic } set_planet_flag = epoch_ii_conclusion_feature_added root = { set_country_flag = epoch_ii_legacy_scar_feature_found } }
			else_if = { limit = { has_modifier = epoch_core_permanent_scar_hive } if = { limit = { NOT = { has_deposit = d_epoch_conclusion_scar_hive } } add_deposit = d_epoch_conclusion_scar_hive } set_planet_flag = epoch_ii_conclusion_feature_added root = { set_country_flag = epoch_ii_legacy_scar_feature_found } }
			else_if = { limit = { has_modifier = epoch_core_permanent_scar_machine } if = { limit = { NOT = { has_deposit = d_epoch_conclusion_scar_machine } } add_deposit = d_epoch_conclusion_scar_machine } set_planet_flag = epoch_ii_conclusion_feature_added root = { set_country_flag = epoch_ii_legacy_scar_feature_found } }
		}
		if = { limit = { has_country_flag = epoch_ii_legacy_scar_feature_found } remove_country_flag = epoch_ii_legacy_scar_feature_not_applicable }
		else = { set_country_flag = epoch_ii_legacy_scar_feature_not_applicable }
		set_country_flag = epoch_ii_legacy_scar_features_migrated
		if = { limit = { has_country_flag = epoch_ii_nullwarden_retreat_pending } epoch_ii_disengage_nullwarden_fleet = yes }
		if = { limit = { NOT = { has_country_flag = epoch_ii_invitation_clock_started } } epoch_ii_start_invitation_clock = yes }
		else_if = {
			limit = { NOR = { has_country_flag = epoch_ii_invited has_country_flag = epoch_ii_invitation_clock has_country_flag = epoch_ii_invite_recovery_armed } }
			# A lost delayed event is recovered only after the persisted due timer
			# has expired, so recovery can be late but can never start early.
			set_timed_country_flag = { flag = epoch_ii_invite_recovery_armed days = 30 }
			country_event = { id = epoch_ii.1 days = 1 }
		}
	}
}
