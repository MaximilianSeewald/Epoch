# situations, and invitation events. No global country iteration is used.
country_event = {
	id = epoch_ii.0
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_country_type = default
		has_country_flag = epoch_core_phase_i_concluded
		has_country_flag = epoch_core_phase_ii_ready
		NOT = { has_country_flag = epoch_core_campaign_opted_out }
	}
	immediate = {
		if = {
			limit = { NOT = { has_country_flag = epoch_ii_contamination_migrated } }
			if = {
				limit = { exists = event_target:epoch_core_world@root }
					event_target:epoch_core_world@root = {
					if = { limit = { root = { OR = { has_country_flag = epoch_core_contamination_worsened has_country_flag = epoch_core_contamination_neglected has_country_flag = epoch_core_contamination_study_failed has_country_flag = epoch_core_contamination_spread_recorded } } } add_deposit = d_epoch_contamination_worsened }
					else_if = { limit = { root = { has_country_flag = epoch_core_contamination_contained } } add_deposit = d_epoch_contamination_contained }
					else_if = { limit = { root = { has_country_flag = epoch_core_contamination_study_complete } } add_deposit = d_epoch_contamination_studied }
					else = { add_deposit = d_epoch_contamination_active }
					set_planet_flag = epoch_ii_contamination_feature_added
				}
				set_country_flag = epoch_ii_contamination_site_recorded
			}
			set_country_flag = epoch_ii_contamination_migrated
		}
		if = {
			limit = { NOT = { has_country_flag = epoch_ii_capital_migrated } }
			if = {
				limit = { exists = event_target:epoch_ii_conclusion_capital@root }
				event_target:epoch_ii_conclusion_capital@root = {
					if = { limit = { root = { is_machine_empire = yes } } add_deposit = d_epoch_conclusion_scar_machine }
					else_if = { limit = { root = { is_hive_empire = yes } } add_deposit = d_epoch_conclusion_scar_hive }
					else = { add_deposit = d_epoch_conclusion_scar_organic }
					set_planet_flag = epoch_ii_conclusion_feature_added
				}
				set_country_flag = epoch_ii_capital_scar_recorded
			}
			else_if = {
				limit = { exists = capital_scope }
				capital_scope = {
					if = { limit = { has_modifier = epoch_core_permanent_scar_organic } add_deposit = d_epoch_conclusion_scar_organic set_planet_flag = epoch_ii_conclusion_feature_added }
					else_if = { limit = { has_modifier = epoch_core_permanent_scar_hive } add_deposit = d_epoch_conclusion_scar_hive set_planet_flag = epoch_ii_conclusion_feature_added }
					else_if = { limit = { has_modifier = epoch_core_permanent_scar_machine } add_deposit = d_epoch_conclusion_scar_machine set_planet_flag = epoch_ii_conclusion_feature_added }
				}
			}
			set_country_flag = epoch_ii_capital_migrated
		}
		if = {
			limit = { NOT = { has_country_flag = epoch_ii_legacy_scar_features_migrated } }
			every_planet = {
				if = { limit = { has_modifier = epoch_core_permanent_scar_organic NOT = { has_planet_flag = epoch_ii_conclusion_feature_added } } add_deposit = d_epoch_conclusion_scar_organic set_planet_flag = epoch_ii_conclusion_feature_added }
				else_if = { limit = { has_modifier = epoch_core_permanent_scar_hive NOT = { has_planet_flag = epoch_ii_conclusion_feature_added } } add_deposit = d_epoch_conclusion_scar_hive set_planet_flag = epoch_ii_conclusion_feature_added }
				else_if = { limit = { has_modifier = epoch_core_permanent_scar_machine NOT = { has_planet_flag = epoch_ii_conclusion_feature_added } } add_deposit = d_epoch_conclusion_scar_machine set_planet_flag = epoch_ii_conclusion_feature_added }
			}
			set_country_flag = epoch_ii_legacy_scar_features_migrated
		}
		if = {
			limit = { NOT = { has_country_flag = epoch_ii_wait_scheduled } }
			set_country_flag = epoch_ii_wait_scheduled
			country_event = { id = epoch_ii.1 days = 1780 random_days = 180 }
		}
		else_if = {
			limit = { has_country_flag = epoch_ii_wait_scheduled NOT = { has_country_flag = epoch_ii_invited } }
			# Yearly one-shot recovery if a delayed event was dropped by a damaged
			# save transition. The invite flag makes every queued copy harmless.
			country_event = { id = epoch_ii.1 days = 366 random_days = 180 }
		}
	}
}
