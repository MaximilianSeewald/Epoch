# One colony/system-local opportunity pass, invoked on surveys, colonization,
# system gain and yearly only while the early window is open.
country_event = {
 id = epoch_echoes.20
 hide_window = yes
 is_triggered_only = yes
 trigger = { epoch_echoes_early_campaign = yes epoch_echoes_pending_discovery = yes }
 immediate = {
  epoch_echoes_select_stories = yes
  if = { limit = {
   has_country_flag = epoch_echoes_unquiet_selected
   NOT = { has_country_flag = epoch_echoes_unquiet_started }
   NOT = { has_country_flag = epoch_echoes_unquiet_delay }
   NOT = { has_country_flag = epoch_echoes_discovery_spacing }
   any_owned_planet = { epoch_echoes_safe_colony = yes }
  }
   random_owned_planet = { limit = { epoch_echoes_safe_colony = yes }
    set_planet_flag = epoch_echoes_unquiet_active
    set_planet_flag = epoch_echoes_unquiet_for@root
    save_global_event_target_as = epoch_echoes_unquiet_world@root
    if = { limit = { owner = { is_hive_empire = yes } } add_modifier = { modifier = epoch_echoes_unquiet_hive days = -1 } }
    else_if = { limit = { owner = { is_robot_empire = yes } } add_modifier = { modifier = epoch_echoes_unquiet_machine days = -1 } }
    else = { add_modifier = { modifier = epoch_echoes_unquiet_organic days = -1 } }
   }
   set_country_flag = epoch_echoes_unquiet_started
   set_timed_country_flag = { flag = epoch_echoes_discovery_spacing days = 120 }
   country_event = { id = epoch_echoes.10 days = 1 }
  }
  if = { limit = {
   has_country_flag = epoch_echoes_relay_selected
   NOT = { has_country_flag = epoch_echoes_relay_started }
   NOT = { has_country_flag = epoch_echoes_relay_delay }
   NOT = { has_country_flag = epoch_echoes_discovery_spacing }
  }
   random_owned_starbase = {
    limit = { solar_system = { any_system_planet = { epoch_echoes_space_site = yes is_surveyed = { who = root status = yes } } } }
    # Reserve the empire's sole construction attempt before creating any object.
    root = {
     set_country_flag = epoch_echoes_relay_started
     set_variable = { which = epoch_echoes_relay_retries value = 0 }
     set_variable = { which = epoch_echoes_relay_age value = 0 }
    }
    solar_system = {
     save_global_event_target_as = epoch_echoes_relay_system@root
     random_system_planet = { limit = { epoch_echoes_space_site = yes is_surveyed = { who = root status = yes } }
      save_global_event_target_as = epoch_echoes_relay_site@root
      set_planet_flag = epoch_echoes_reserved
     }
     spawn_megastructure = { type = epoch_echoes_relay_dormant owner = root planet = event_target:epoch_echoes_relay_site@root }
     random_system_megastructure = { limit = { is_megastructure_type = epoch_echoes_relay_dormant NOT = { has_megastructure_flag = epoch_echoes_claimed } }
      set_megastructure_flag = epoch_echoes_claimed
      set_megastructure_flag = epoch_echoes_relay_for@root
      save_global_event_target_as = epoch_echoes_relay_object@root
     }
    }
   }
   if = { limit = { exists = event_target:epoch_echoes_relay_object@root }
    set_variable = { which = epoch_echoes_relay_retries value = 0 }
    set_variable = { which = epoch_echoes_relay_age value = 0 }
    set_country_flag = epoch_echoes_relay_started
    set_timed_country_flag = { flag = epoch_echoes_discovery_spacing days = 120 }
    epoch_echoes_register_relay_poi = yes
    country_event = { id = epoch_echoes.31 days = 1 }
   }
   else_if = { limit = { has_country_flag = epoch_echoes_relay_started }
    epoch_echoes_close_relay = yes
   }
  }
 }
}
