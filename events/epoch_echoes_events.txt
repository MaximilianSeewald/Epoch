# Survey callback: at most one roll per empire per survey. Only already
# surveyed, reachable systems become story targets.
ship_event = {
	id = epoch_echoes.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = owner
		owner = {
			epoch_echoes_early_campaign = yes
			NOT = { has_country_flag = epoch_echoes_ghost_started }
			NOT = { has_country_flag = epoch_echoes_ghost_closed }
			NOT = { has_country_flag = epoch_echoes_ghost_trace_complete }
			NOT = { has_country_flag = epoch_echoes_relay_started }
		}
	}
	immediate = {
		if = { limit = { NOT = { exists = event_target:epoch_echoes_story_site@root.owner } }
			from = { solar_system = { save_global_event_target_as = epoch_echoes_story_system@root.owner random_system_planet = { save_global_event_target_as = epoch_echoes_story_site@root.owner } } }
		}
		random = {
			chance = 12
			from = {
				solar_system = {
					save_global_event_target_as = epoch_echoes_story_system@root.owner
					random_system_planet = { save_global_event_target_as = epoch_echoes_story_site@root.owner }
				}
			}
			owner = {
				if = { limit = { exists = event_target:epoch_echoes_story_site@root }
					set_country_flag = epoch_echoes_ghost_started
					epoch_echoes_register_relay_poi = yes
					country_event = { id = epoch_echoes.2 days = 1 }
				}
			}
		}
		if = {
			limit = {
				owner = {
					epoch_echoes_early_campaign = yes
					NOT = { has_country_flag = epoch_echoes_unquiet_started }
					any_owned_planet = { epoch_echoes_safe_colony = yes }
				}
			}
			random = {
				chance = 10
				owner = {
					set_country_flag = epoch_echoes_unquiet_started
					random_owned_planet = {
						limit = { epoch_echoes_safe_colony = yes }
						set_planet_flag = epoch_echoes_unquiet_active
						if = { limit = { owner = { is_hive_empire = yes } } add_modifier = { modifier = epoch_echoes_unquiet_hive days = 3600 } }
						else_if = { limit = { owner = { is_robot_empire = yes } } add_modifier = { modifier = epoch_echoes_unquiet_machine days = 3600 } }
						else = { add_modifier = { modifier = epoch_echoes_unquiet_organic days = 3600 } }
						save_global_event_target_as = epoch_echoes_unquiet_world@root.owner
					}
					if = { limit = { exists = event_target:epoch_echoes_story_system@root } epoch_echoes_register_relay_poi = yes }
					if = { limit = { exists = event_target:epoch_echoes_unquiet_world@root.owner } country_event = { id = epoch_echoes.10 days = 1 } }
				}
			}
		}
	}
}

# One owner-local yearly repair for a permanently declined personal inquiry.
# Lost colonies retain only a finite penalty modifier, which expires naturally.
country_event = {
	id = epoch_echoes.22
	hide_window = yes
	is_triggered_only = yes
	trigger = { has_country_flag = epoch_core_campaign_opted_out OR = { has_country_flag = epoch_echoes_unquiet_started has_country_flag = epoch_echoes_ghost_started has_country_flag = epoch_echoes_relay_started } NOT = { has_country_flag = epoch_echoes_decline_cleaned } }
	immediate = {
		every_owned_planet = {
			limit = { has_planet_flag = epoch_echoes_unquiet_active }
			remove_modifier = epoch_echoes_unquiet_organic
			remove_modifier = epoch_echoes_unquiet_hive
			remove_modifier = epoch_echoes_unquiet_machine
			remove_planet_flag = epoch_echoes_unquiet_active
			set_planet_flag = epoch_echoes_unquiet_closed
		}
		if = { limit = { has_country_flag = epoch_echoes_ghost_started NOT = { has_country_flag = epoch_echoes_ghost_trace_complete } }
			abort_special_project = { type = EPOCH_ECHOES_GHOST_TRACE }
			abort_special_project = { type = EPOCH_ECHOES_GHOST_CAUTIOUS }
			abort_special_project = { type = EPOCH_ECHOES_GHOST_FORCED }
			abort_special_project = { type = EPOCH_ECHOES_GHOST_STABILIZE }
			set_country_flag = epoch_echoes_ghost_closed
		}
		if = { limit = { has_country_flag = epoch_echoes_relay_started NOT = { has_country_flag = epoch_echoes_relay_resolved } }
			abort_special_project = { type = EPOCH_ECHOES_RELAY_SURVEY }
			abort_special_project = { type = EPOCH_ECHOES_RELAY_DEFUSE }
			abort_special_project = { type = EPOCH_ECHOES_RELAY_DEFUSE_AIDED }
			set_country_flag = epoch_echoes_relay_resolved
			remove_country_flag = epoch_echoes_relay_combat_pending
			if = { limit = { exists = event_target:epoch_echoes_sentinel_country@root }
				event_target:epoch_echoes_sentinel_country@root = { every_owned_fleet = { limit = { has_fleet_flag = epoch_echoes_relay_guard@root } destroy_fleet = this } }
			}
		}
		if = { limit = { has_country_flag = epoch_core_campaign_opted_out }
			if = { limit = { has_point_of_interest = { poi = epoch_echoes_relay_site } } remove_point_of_interest = epoch_echoes_relay_site }
			if = { limit = { has_event_chain = epoch_echoes_side_stories } end_event_chain = epoch_echoes_side_stories }
		}
		set_country_flag = epoch_echoes_decline_cleaned
	}
}
